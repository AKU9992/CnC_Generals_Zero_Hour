cmake_minimum_required(VERSION 3.24)
project(GeneralsNative12 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native renderer requires Windows AMD64")
endif()
add_library(generals_native12 STATIC NativeDevice12.cpp)
target_compile_features(generals_native12 PUBLIC cxx_std_17)
target_compile_definitions(generals_native12 PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(generals_native12 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(generals_native12 PUBLIC d3d12 dxgi d3dcompiler)
if(MSVC)
  target_compile_options(generals_native12 PRIVATE /W4 /WX)
endif()
add_executable(NativeRenderer12Tests ../../tools/native/NativeRenderer12Tests.cpp)
target_link_libraries(NativeRenderer12Tests PRIVATE generals_native12 user32)
if(MSVC)
  target_compile_options(NativeRenderer12Tests PRIVATE /W4 /WX)
endif()
enable_testing()
add_test(NAME native12_gpu COMMAND NativeRenderer12Tests)
set_tests_properties(native12_gpu PROPERTIES TIMEOUT 60)
set(STREAMLINE_INCLUDE_DIR "" CACHE PATH "Verified pinned Streamline SDK include directory")
if(STREAMLINE_INCLUDE_DIR)
  add_library(generals_native_scene12 STATIC NativeScene12.cpp)
  target_include_directories(generals_native_scene12 PUBLIC ${STREAMLINE_INCLUDE_DIR})
  target_link_libraries(generals_native_scene12 PUBLIC generals_native12)
  add_executable(NativeDlss12Tests ../../tools/native/NativeDlss12Tests.cpp)
  target_include_directories(NativeDlss12Tests PRIVATE ${STREAMLINE_INCLUDE_DIR})
  target_link_libraries(NativeDlss12Tests PRIVATE generals_native12 user32)
  if(MSVC)
    target_compile_options(NativeDlss12Tests PRIVATE /W4 /WX)
    target_compile_options(generals_native_scene12 PRIVATE /W4 /WX)
  endif()
  add_test(NAME native12_dlss COMMAND NativeDlss12Tests)
  set_tests_properties(native12_dlss PROPERTIES TIMEOUT 120)
  add_executable(NativeScene12Tests ../../tools/native/NativeScene12Tests.cpp)
  target_link_libraries(NativeScene12Tests PRIVATE generals_native_scene12 user32)
  if(MSVC)
    target_compile_options(NativeScene12Tests PRIVATE /W4 /WX)
  endif()
  add_test(NAME native12_scene COMMAND NativeScene12Tests)
  set_tests_properties(native12_scene PROPERTIES TIMEOUT 120)
endif()
